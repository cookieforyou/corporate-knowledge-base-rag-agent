## 7.1 PostgreSQL 核心表设计

```sql
-- ============================================
-- PostgreSQL 18 DDL（所有建表语句后执行独立 CREATE INDEX）
-- ============================================

-- 1. 文档主表
-- v2.25 扩展（簇⑥ C1 增量重入库）：version 列 + status 枚举增 REINDEXING
-- 存量库升级 DDL：
--   ALTER TABLE kb_document ADD COLUMN IF NOT EXISTS version INT NOT NULL DEFAULT 1;
CREATE TABLE kb_document (
    id              VARCHAR(36) PRIMARY KEY,
    tenant_id       VARCHAR(36) NOT NULL,
    name            VARCHAR(255) NOT NULL,
    original_name   VARCHAR(500),
    type            VARCHAR(20) NOT NULL,           -- PDF, DOCX, MD, TXT
    size            BIGINT,
    oss_path        VARCHAR(500),
    status          VARCHAR(20) DEFAULT 'UPLOADING', -- UPLOADING/PARSING/REINDEXING/SUCCESS/FAILED
    parse_route     VARCHAR(20),                   -- NATIVE, DEEP, OCR
    page_count      INT,
    table_count     INT,
    image_count     INT,
    chunk_count     INT,
    error_message   TEXT,
    version         INT NOT NULL DEFAULT 1,        -- 簇⑥ C1：重入库成功 +1
    created_by      VARCHAR(50),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_tenant_status ON kb_document (tenant_id, status);
CREATE INDEX idx_created_at ON kb_document (created_at);

-- 2. 章节表
CREATE TABLE kb_section (
    id              VARCHAR(36) PRIMARY KEY,
    doc_id          VARCHAR(36) NOT NULL REFERENCES kb_document(id) ON DELETE CASCADE,
    parent_id       VARCHAR(36),
    title           VARCHAR(500),
    level           INT DEFAULT 1,
    order_index     INT,
    page_start      INT,
    page_end        INT
);
CREATE INDEX idx_doc_section ON kb_section (doc_id, order_index);

-- 3. 切分块表（核心业务表）
CREATE TABLE kb_chunk (
    id              VARCHAR(36) PRIMARY KEY,
    doc_id          VARCHAR(36) NOT NULL REFERENCES kb_document(id) ON DELETE CASCADE,
    section_id      VARCHAR(36),
    chunk_index     INT NOT NULL,
    content         TEXT NOT NULL,
    original_content TEXT,
    page_num        INT,
    token_count     INT,
    metadata        JSONB DEFAULT '{}',
    chunk_type      VARCHAR(20) DEFAULT 'TEXT',    -- TEXT, TABLE, IMAGE
    vector_id       VARCHAR(100),
    is_deleted      BOOLEAN DEFAULT FALSE,
    -- created_at：ORM 侧 @Column(updatable=false)（v2.26，簇⑥ C1 E2E 修复）——
    -- 蓝绿重入库同 ID merge 保留原创建时间，仅 INSERT 写入
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_doc_chunk ON kb_chunk (doc_id, chunk_index);
CREATE INDEX idx_vector_id ON kb_chunk (vector_id);
CREATE INDEX idx_chunk_type ON kb_chunk (chunk_type);
-- PG 全文检索 GIN 索引（admin 管理后台查询用，非检索主路径）
CREATE INDEX idx_ft_content ON kb_chunk USING GIN (to_tsvector('simple', content));

-- 4. 会话表
CREATE TABLE kb_session (
    id              VARCHAR(36) PRIMARY KEY,
    tenant_id       VARCHAR(36) NOT NULL,
    user_id         VARCHAR(50) NOT NULL,
    title           VARCHAR(255),
    knowledge_base  VARCHAR(100),
    status          VARCHAR(20) DEFAULT 'ACTIVE',
    message_count   INT DEFAULT 0,
    total_tokens    BIGINT DEFAULT 0,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_user_session ON kb_session (user_id, updated_at DESC);

-- 5. 消息表
CREATE TABLE kb_message (
    id              VARCHAR(36) PRIMARY KEY,
    session_id      VARCHAR(36) NOT NULL REFERENCES kb_session(id) ON DELETE CASCADE,
    role            VARCHAR(20) NOT NULL,
    content         TEXT,
    citations       JSONB,
    token_usage     JSONB,
    metadata        JSONB DEFAULT '{}',
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_session_msg ON kb_message (session_id, created_at);

-- 6. 审计日志表
-- v2.10 扩展（2026-08-05，3.12 审计落地，双链路时代补列）：
-- mode（rag|tool）/ status（SUCCESS/REJECTED/ERROR）/ error_code / tool_calls(JSONB)
-- 存量库升级：ALTER TABLE kb_audit_log ADD COLUMN IF NOT EXISTS <列名> <类型>;（四条）
-- 注意：ddl-auto=validate，缺列启动失败——升级须先于新版本部署执行
CREATE TABLE kb_audit_log (
    id              BIGSERIAL PRIMARY KEY,
    trace_id        VARCHAR(100),
    session_id      VARCHAR(36),
    user_id         VARCHAR(50),
    tenant_id       VARCHAR(36),
    mode            VARCHAR(10),
    query_text      TEXT NOT NULL,
    rewritten_query TEXT,
    retrieval_type  VARCHAR(30),
    retrieved_chunks JSONB,
    reranked_chunks  JSONB,
    final_answer    TEXT,
    tool_calls      JSONB,
    model_name      VARCHAR(100),
    latency_ms      INT,
    token_usage     JSONB,
    status          VARCHAR(20),
    error_code      VARCHAR(50),
    feedback        VARCHAR(10),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
CREATE INDEX idx_audit_trace ON kb_audit_log (trace_id);
CREATE INDEX idx_audit_session ON kb_audit_log (session_id);
CREATE INDEX idx_audit_user ON kb_audit_log (user_id);
CREATE INDEX idx_audit_created ON kb_audit_log (created_at DESC);

-- 7. 用户反馈表
CREATE TABLE kb_feedback (
    id              VARCHAR(36) PRIMARY KEY,
    message_id      VARCHAR(36) NOT NULL REFERENCES kb_message(id),
    audit_log_id    BIGINT REFERENCES kb_audit_log(id),
    user_id         VARCHAR(50),
    rating          VARCHAR(10) NOT NULL,
    expected_answer TEXT,
    feedback_tags   JSONB,
    resolved        BOOLEAN DEFAULT FALSE,
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 8. Prompt 模板表
CREATE TABLE kb_prompt_template (
    id              VARCHAR(36) PRIMARY KEY,
    name            VARCHAR(100) NOT NULL,
    version         VARCHAR(20) NOT NULL,
    template_text   TEXT NOT NULL,
    variables       JSONB,
    status          VARCHAR(20) DEFAULT 'DRAFT',
    ab_test_group   VARCHAR(20),
    created_by      VARCHAR(50),
    created_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uk_name_version UNIQUE (name, version)
);
```

## 7.2 pgvector 向量表设计（pgvector 模式）

```sql
-- pgvector 模式下，向量存储在 PostgreSQL 同一实例中
-- 表由 Spring AI PgVectorStore 自动创建（initialize-schema=true），也可手动执行

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE IF NOT EXISTS kb_embeddings (
    id          VARCHAR(36) PRIMARY KEY,           -- 对应 kb_chunk.id
    embedding   vector(1024),                      -- 向量（维度与 EmbeddingModel 一致）
    content     TEXT,                               -- Chunk 原文冗余（可选，便于调试）
    metadata    JSONB DEFAULT '{}',                -- chunk_id / doc_id / tenant_id / chunk_type
    created_at  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 向量索引（HNSW，查询性能最优）
CREATE INDEX IF NOT EXISTS idx_embedding_hnsw
    ON kb_embeddings USING hnsw (embedding vector_cosine_ops);

-- 元数据查询索引
CREATE INDEX IF NOT EXISTS idx_emb_metadata
    ON kb_embeddings USING GIN (metadata);
```

> **v2.28 修正（2026-08-13，簇⑥ D3 集成测试实证）——pgvector 接线两则**：
> ① **idType 钉 TEXT**：kb_embeddings.id 为 VARCHAR(36) 融合键（chunk 确定性 nameUUID
> 字符串）；PgVectorStore 默认 idType=UUID 时 insert 走 uuid→varchar 隐式转换可写入，
> 但 delete 以 uuid 参数匹配 varchar 列**类型失配静默不删**（清理组件尽力而为吞异常，
> 表象为向量残留）——VectorStoreConfig 装配钉 `.idType(PgIdType.TEXT)`。
> ② **schema.sql 自包含**：kb-domain schema.sql 补 `CREATE EXTENSION IF NOT EXISTS vector`
> （原口径「superuser 先行单独执行」收入文件）——Testcontainers init script 以 postgres
> 超级用户直接执行该文件建全套 schema，测试 DDL 与生产单一来源零漂移。

**pgvector 模式配置**：

```yaml
kb:
  vector-store:
    provider: pgvector
    pgvector:
      table-name: kb_embeddings
      schema-name: public
      dimensions: 1024
      distance-type: COSINE_DISTANCE
      index-type: HNSW
      initialize-schema: true
```

- 向量表与主数据表同库，通过 `id = kb_chunk.id` 外键关联
- pgvector 的 HNSW 索引在小规模（< 10 万向量）下性能与 Milvus 接近
- 无需额外中间件，减少运维复杂度

## 7.6 Schema 迁移版本化：Flyway（v2.54 新增，Phase 4 簇⑥ 4.11）

> 背景：Phase 1-3 至安全专项期间 schema 演进为「schema.sql 全量快照 + ECS 手工 ALTER（ddl-auto=validate 校验兜底）」形态——第 18 章「DB 迁移版本化 100%」验收缺口，v2.29 复审定案引入 Flyway 归生产加固簇。

**依赖与装配**：kb-domain 引入 `spring-boot-starter-flyway`（Boot 4.x 新 starter，传递 spring-boot-flyway → flyway-core 12.4.0，Boot 4.1 BOM 管理）+ `flyway-database-postgresql` **显式声明**——Flyway 10+ 数据库模块拆分，PG18 缺该模块即 "Unsupported Database: PostgreSQL 18" 启动失败（Boot #49012 实证登记）。迁移脚本归 schema 事实源模块 kb-domain `src/main/resources/db/migration/`（Flyway 默认 locations 零配置命中），kb-api 运行时宿主经 Boot 自动配置在 Hibernate 校验之前触发迁移。

**基线策略（ECS 现网库存量表）**：`spring.flyway.baseline-on-migrate=true` + `baseline-version=0`（application-infra.yml，`FLYWAY_ENABLED`/`FLYWAY_BASELINE_ON_MIGRATE` env 阀门保留）——

- 存量库（表已全在、无 flyway_schema_history）：首跑插 baseline 行（version=0）后应用 V1；V1 全 DDL 幂等形态（CREATE EXTENSION/TABLE/INDEX IF NOT EXISTS）逐条 no-op，**零结构变更**，仅登记 history 两行；
- 空库（全新环境）：检测空 schema 跳过 baseline，V1 直建全套。

**V1 基线脚本**：`V1__baseline_schema.sql` = `schema.sql` v2.53.1 全量快照（pgvector 扩展 + 10 业务表 + 全套索引）。CREATE EXTENSION IF NOT EXISTS 对已装扩展先于权限检查短路返回（PG 源 extension.c 存在性检查在前，NOTICE 即返）——非 superuser 的应用连接用户在存量库上执行安全（ECS 扩展已预装）。

**同源双写纪律（钉死）**：schema.sql 保留——kb-eval Testcontainers IT init script 专用（33 IT 复用，超级用户执行）。后续任何 schema 变更：① 先新增 Flyway 迁移脚本 `V(N+1)__xxx.sql`；② 同步合入 schema.sql 全量快照。表集一致性由 kb-domain `SchemaDualSourceConsistencyTest` 单测兜底（双源 CREATE TABLE 集合比对，漏同步即红灯）；列级漂移由 IT 的 ddl-auto=validate 兜底。

**上下文边界（实证定案）**：

| 上下文 | Flyway 形态 | 理由 |
|---|---|---|
| kb-api 生产/开发 | 开（缺省） | 迁移权唯一归属，启动期先于 Hibernate validate |
| kb-eval 主上下文 | 显式关（application.yml） | 度量工具不向目标库写 baseline 登记 |
| kb-eval IT | DynamicPropertySource 显式重开 | init script 建库 = 非空库无 history 形态，与 ECS 现网首跑同构——baseline 路径持续回归保险 |

**纪律**：ddl-auto=validate 保留（Flyway 管演进 + Hibernate 校验 Entity↔DDL 一致，双保险）；历史「存量库升级 DDL」ALTER 注释（各表节内）自 v2.54 起由 Flyway 迁移脚本取代，不再手工 ALTER。
